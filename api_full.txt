     3	/**
     4	 * Tell the server a planet started farming. The server uses this to
     5	 * schedule the "Farm full" Telegram notification 24h later. Fire-and-forget
     6	 * — failures are silently ignored so a network blip never blocks gameplay.
     7	 */
     8	export function notifyFarmStart(telegramId: string, planetId: string, planetType: string, isWhite = false): void {
     9	  if (!telegramId || !planetId) return;
    10	  fetch(`${API_BASE}/farm/start`, {
    11	    method: "POST",
    12	    headers: { "Content-Type": "application/json" },
    13	    body: JSON.stringify({ telegramId, planetId, planetType, isWhite }),
    14	    keepalive: true,
    15	  }).catch(() => { /* ignore */ });
    16	}

    61	// Returns the server's current epoch ms, or null if it can't be obtained.
    62	// We never silently fall back to Date.now() because callers use this value
    63	// to *detect* clock-tampering — substituting the local clock on failure
    64	// would defeat the very check (e.g. the stardust spawn anti-tamper schedule
    65	// would treat the local clock as authoritative whenever /time is briefly
    66	// unreachable, re-opening the exploit). Callers must handle null explicitly
    67	// (typically: skip persistence, or skip honouring a saved timestamp).
    68	export async function fetchServerTime(): Promise<number | null> {
    69	  try {
    70	    const res = await fetch(`${API_BASE}/time?t=${Date.now()}`, { cache: "no-store" });
    71	    if (!res.ok) return null;
    72	    const data = await res.json();
    73	    return typeof data.serverTime === "number" ? data.serverTime : null;
    74	  } catch {
    75	    return null;
    76	  }
    77	}

   172	  telegramId: string;
   173	  firstName?: string | null;
   174	  username?: string | null;
   175	  zoomBalance: number;
   176	  tonBalance?: number;
   177	  clientEpoch?: number;
   178	}): Promise<{ zoomBalance: number; tonBalance: number; balanceEpoch: number }> {
   179	  const fallbackTon = typeof params.tonBalance === "number" ? params.tonBalance : 0;
   180	  try {
   181	    const res = await fetch(`${API_BASE}/balance/sync`, {
   182	      method: "POST",
   183	      headers: { "Content-Type": "application/json" },
   184	      body: JSON.stringify(params),
   185	    });
   186	    if (!res.ok) return { zoomBalance: params.zoomBalance, tonBalance: fallbackTon, balanceEpoch: params.clientEpoch ?? 0 };
   187	    const data = await res.json();
   188	    return {
   189	      zoomBalance: typeof data.zoomBalance === "number" ? data.zoomBalance : params.zoomBalance,
   190	      tonBalance: typeof data.tonBalance === "number" ? data.tonBalance : fallbackTon,
   191	      balanceEpoch: typeof data.balanceEpoch === "number" ? data.balanceEpoch : (params.clientEpoch ?? 0),
   192	    };
   193	  } catch {
   194	    return { zoomBalance: params.zoomBalance, tonBalance: fallbackTon, balanceEpoch: params.clientEpoch ?? 0 };
   195	  }

   222	/**
   223	 * Push the current SUN cycle to the server so it persists across
   224	 * localStorage loss. Server merges with GREATEST per field — replaying

   227	 * Fire-and-forget at the call site: the local state is already updated
   228	 * optimistically. A network failure here just means the user's other

   230	 * away — no data loss.

   232	export async function syncSunCycle(params: {

   240	      method: "POST",
   241	      headers: { "Content-Type": "application/json" },
   242	      body: JSON.stringify(params),
   243	    });
   244	  } catch { /* fire-and-forget */ }
   245	}

   775	export async function pollTxnUntilFinal(txnId: number, opts: { maxMs?: number; intervalMs?: number } = {}): Promise<{ status: string; itemName?: string } | null> {
   776	  const maxMs = opts.maxMs ?? 180_000;
   777	  const intervalMs = opts.intervalMs ?? 4_000;
   778	  const start = Date.now();
   779	  while (Date.now() - start < maxMs) {
   780	    const s = await fetchTxnStatus(txnId);
   781	    if (s && (s.status === "completed" || s.status === "failed")) return s;
   782	    await new Promise((r) => setTimeout(r, intervalMs));
   783	  }
   784	  return null;
   785	}

   989	export function openMarketActivityStream(onSale: (sale: MarketSale) => void): () => void {
   990	  let es: EventSource | null = null;
   991	  let closed = false;
   992	  let retry: ReturnType<typeof setTimeout> | null = null;
   993	
   994	  const connect = () => {
   995	    if (closed) return;
   996	    try {
   997	      es = new EventSource(`${API_BASE}/market/activity/stream`);
   998	      es.addEventListener("sale", (e) => {
   999	        try { onSale(JSON.parse((e as MessageEvent).data)); } catch { /* */ }
  1000	      });
  1001	      es.onerror = () => {
  1002	        try { es?.close(); } catch { /* */ }
  1003	        es = null;
  1004	        if (!closed) retry = setTimeout(connect, 3000);
  1005	      };
  1006	    } catch { if (!closed) retry = setTimeout(connect, 3000); }
  1007	  };
  1008	  connect();
  1009	  return () => {
  1010	    closed = true;
  1011	    if (retry) clearTimeout(retry);
  1012	    try { es?.close(); } catch { /* */ }
  1013	  };
  1014	}

  1353	// ─── Regular planets (FarmPage main grid) — server-side persistence ───
  1354	// Lets the user's planet inventory follow them across devices and survive
  1355	// any localStorage wipe. The server stores the array as opaque JSONB and
  1356	// also mirrors the per-rarity claimed-bonus counters so applyGrants on a
  1357	// fresh device doesn't re-mint bonus planets that were already burned.
  1358	export interface RegularPlanetsState {
  1359	  // `ok` distinguishes a SUCCESSFUL fetch (with possibly empty/missing data)
  1360	  // from a transient FAILURE (network error, 5xx). Callers must NOT enable
  1361	  // their server-write gate when ok is false — otherwise a flaky network
  1362	  // could silently clobber the server inventory with a stale local snapshot.
  1363	  ok: boolean;
  1364	  // True if the user row exists on the server. False both for new users and
  1365	  // for failed fetches (check `ok` first to disambiguate).
  1366	  exists: boolean;
  1367	  // Uses `unknown` here so we don't pull the full Planet type into the
  1368	  // utils layer; the caller (useGameState) re-shapes these into Planet[].
  1369	  planets: Array<Record<string, unknown>>;
  1370	  claimedBonusBasic: number;
  1371	  claimedBonusRare: number;
  1372	  claimedBonusEpic: number;
  1373	  claimedBonusGold: number;
  1374	  claimedBonusV1: number;
  1375	}

  1384	    claimedBonusBasic: 0,
