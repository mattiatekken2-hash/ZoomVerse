     5	// Server-authoritative clock: every farming/idle-income time check is computed
     6	// against this value, NOT the device clock. Calibrated against /api/server-time
     7	// so a tampered phone clock cannot accelerate ZOOM/TON accrual.
     8	let _serverOffsetMs = 0;
     9	let _serverOffsetReady = false;
    10	
    11	export function serverNow(): number {
    12	  return Date.now() + _serverOffsetMs;
    13	}
    14	
    15	export function isServerClockReady(): boolean {
    16	  return _serverOffsetReady;
    17	}

    19	// Returns the calibrated offset, or null if the server time couldn't be
    20	// obtained. Callers must NOT silently treat null as "0 offset" — that would
    21	// flag the clock as ready while it's actually still on the device's local
    22	// clock, defeating any anti-tamper logic that relies on _serverOffsetReady.
    23	async function calibrateServerOffset(): Promise<number | null> {
    24	  try {
    25	    const t0 = Date.now();
    26	    const serverTime = await fetchServerTime();
    27	    if (serverTime == null) return null;
    28	    const t1 = Date.now();
    29	    const rtt = t1 - t0;
    30	    return serverTime - (t0 + rtt / 2);
    31	  } catch {
    32	    return null;
    33	  }
    34	}
    35	
    36	async function refreshServerOffset(): Promise<void> {
    37	  try {
    38	    const offset = await calibrateServerOffset();
    39	    if (offset == null) return; // /time unreachable — keep last known state.
    40	    // Sanity check: if RTT-noise produced something insane, ignore it.
    41	    if (Number.isFinite(offset) && Math.abs(offset) < 365 * 24 * 3_600_000) {
    42	      _serverOffsetMs = offset;
    43	      _serverOffsetReady = true;
    44	    }
    45	  } catch { /* keep last known offset */ }
    46	}

   379	function getTelegramContext(): { telegramId: string | null; startParam: string | null; firstName: string | null; username: string | null } {
   380	  try {
   381	    const webApp = (window as unknown as { Telegram?: { WebApp?: { initDataUnsafe?: { user?: { id?: number; first_name?: string; username?: string }; start_param?: string }; initData?: string } } }).Telegram?.WebApp;
   382	    const unsafe = webApp?.initDataUnsafe;
   383	    const telegramId = unsafe?.user?.id ? String(unsafe.user.id) : null;
   384	    const firstName = unsafe?.user?.first_name ?? null;
   385	    const username = unsafe?.user?.username ?? null;
   386	
   387	    let startParam: string | null = unsafe?.start_param || null;
   388	
   389	    if (!startParam && webApp?.initData) {
   390	      try {
   391	        const params = new URLSearchParams(webApp.initData);
   392	        startParam = params.get("start_param");
   393	      } catch { /**/ }
   394	    }
   395	
   396	    if (!startParam) {
   397	      startParam = localStorage.getItem("zoom-start-param");
   398	    }
   399	
   400	    return { telegramId, startParam, firstName, username };
   401	  } catch {
   402	    return { telegramId: null, startParam: null, firstName: null, username: null };
   403	  }
   404	}

   474	function loadState(): GameState {
   475	  const { telegramId, startParam, firstName: _firstName } = getTelegramContext();
   476	
   477	  try {
   478	    const matchingKey = localStorage.getItem(getStorageKey(telegramId));
   479	    const raw = matchingKey ?? localStorage.getItem(STORAGE_KEY);
   480	    if (raw) {
   481	      const parsed = JSON.parse(raw) as GameState;
   482	      if (parsed.version === STATE_VERSION) {
   483	        if (telegramId && parsed.telegramId !== telegramId) {
   484	          _lastLoadWasFresh = true;
   485	          const referredBy = startParam ? startParam : null;
   486	          return {
   487	            ...INITIAL_STATE,
   488	            referralCode: telegramId,
   489	            telegramId,
   490	            referredBy,
   491	            referralSpeedBonus: referredBy ? 0.10 : 0,
   492	          };
   493	        }
   494	        const base: GameState = {
   495	          ...INITIAL_STATE,
   496	          ...parsed,
   497	          planets: (parsed.planets || []).map(migratePlanet),
   498	          pendingPlanet: parsed.pendingPlanet ? migratePlanet(parsed.pendingPlanet) : null,
   499	          usedRedeemCodes: parsed.usedRedeemCodes || [],
   500	          sun: parsed.sun || null,
   501	          referralSpeedBonus: parsed.referralSpeedBonus ?? 0,
   502	          referredBy: parsed.referredBy ?? null,
   503	          telegramId: parsed.telegramId ?? null,
   504	          claimedBonusSun: parsed.claimedBonusSun ?? false,
   505	          lastFarmingSettledAt: parsed.lastFarmingSettledAt ?? serverNow(),
   506	          claimedMilestones: parsed.claimedMilestones ?? [],
   507	          lastBalanceEpoch: parsed.lastBalanceEpoch ?? 0,
   508	          whiteCollectionBundles: (parsed as unknown as Record<string, unknown>).whiteCollectionBundles as number ?? (parsed.whiteCollectionUnlocked ? 1 : 0),
   509	          claimedWhiteCollectionBundles:
   510	            (parsed as unknown as Record<string, unknown>).claimedWhiteCollectionBundles as number
   511	            ?? ((parsed as unknown as Record<string, unknown>).claimedWhiteCollection ? 1 : 0),
   512	          whitePlanets: (parsed.whitePlanets || []).map(migratePlanet),
   513	          earthCollectionUnlocked: (parsed as unknown as Record<string, unknown>).earthCollectionUnlocked as boolean ?? false,
   514	          earthCollectionBundles: (parsed as unknown as Record<string, unknown>).earthCollectionBundles as number ?? 0,
   515	          claimedEarthCollectionBundles: (parsed as unknown as Record<string, unknown>).claimedEarthCollectionBundles as number ?? 0,
   516	          earthPlanets: ((parsed as unknown as Record<string, unknown>).earthPlanets as Planet[] | undefined ?? []).map(migratePlanet),
   517	          tonBalance: parsed.tonBalance ?? 0,
   518	        };
   519	        const resolvedTelegramId = telegramId || base.telegramId;
   520	        // Only treat as "fresh load" when we did NOT find an entry keyed to the
   521	        // current Telegram user. If matchingKey is null we fell back to a
   522	        // legacy un-keyed entry (or someone else's), so still treat as fresh
   523	        // and let server be authoritative on first sync.
   524	        _lastLoadWasFresh = !matchingKey;
   525	        return {
   526	          ...base,
   527	          telegramId: resolvedTelegramId,
   528	          referralCode: resolvedTelegramId || base.referralCode,
   529	        };
   530	      }
   531	    }

   537	  const referralSpeedBonus = referredBy ? 0.10 : 0;
   538	  const referralCode = telegramId || makeReferralCode();
   539	
   540	  return {
   541	    ...INITIAL_STATE,
   542	    referralCode,
   543	    telegramId,
   544	    referredBy,
   545	    referralSpeedBonus,
   546	  };

   554	function saveState(state: GameState) {
   555	  // Discard any queued idle write — this snapshot is newer and authoritative.
   556	  // Without this, a stale schedulePersist payload (e.g. from a tap a few ms
   557	  // earlier) could fire AFTER us and resurrect items that were just removed
   558	  // (burned planets, sold items, etc.) on the next reload.
   559	  _pendingPersistState = null;
   560	  _persistScheduled = false;
   561	  _writeSeq++;
   562	  _lastSavedAt = Date.now();
   563	  try {
   564	    localStorage.setItem(getStorageKey(state.telegramId), JSON.stringify(state));
   565	  } catch { /**/ }
   566	}

   568	// Non-blocking persistence scheduler. Each call replaces the pending state and
   569	// the actual JSON.stringify + localStorage write happens during the browser's
   570	// idle time (or next animation frame as fallback). This keeps the tap thread
   571	// at 60fps even when state grows large (many planets, feed events, etc).
   572	let _pendingPersistState: GameState | null = null;
   573	let _persistScheduled = false;
   574	type IdleCallback = (cb: () => void, opts?: { timeout?: number }) => number;
   575	const _scheduleIdle: IdleCallback =
   576	  typeof window !== "undefined" && typeof (window as unknown as { requestIdleCallback?: IdleCallback }).requestIdleCallback === "function"
   577	    ? (window as unknown as { requestIdleCallback: IdleCallback }).requestIdleCallback.bind(window)
   578	    : ((cb: () => void) => window.setTimeout(cb, 0)) as IdleCallback;
   579	
   580	function schedulePersist(state: GameState) {
   581	  _pendingPersistState = state;
   582	  if (_persistScheduled) return;
   583	  _persistScheduled = true;
   584	  const seqAtSchedule = _writeSeq;
   585	  _scheduleIdle(() => {
   586	    _persistScheduled = false;
   587	    const s = _pendingPersistState;
   588	    _pendingPersistState = null;
   589	    // If anyone wrote authoritatively while we were queued, the snapshot we
   590	    // captured is potentially stale (e.g. user burned/sold/listed an item
   591	    // between schedule and idle). Skip — the authoritative writer already
   592	    // persisted the truth.
   593	    if (_writeSeq !== seqAtSchedule) return;
   594	    if (s) saveState(s);
   595	  }, { timeout: 200 });
   596	}

   598	// Force-flush pending persist (used on page hide / unload to guarantee writes).
   599	function flushPersist() {
   600	  if (_pendingPersistState) {
   601	    const s = _pendingPersistState;
   602	    _pendingPersistState = null;
   603	    _persistScheduled = false;
   604	    saveState(s);
   605	  }
   606	}

   633	// Called after every /balance/sync response. The server is authoritative
   634	// whenever its epoch is higher than the one we sent — that means an
   635	// authoritative balance change happened on the server (admin mutation,
   636	// Stars/TON purchase credit, marketplace buy/sell, wheel/daily/referral
   637	// reward) since our last sync. In that case we MUST snap the local state
   638	// to the server value (regardless of whether it is higher or lower) so the
   639	// next outgoing sync doesn't echo back the stale value and overwrite the
   640	// authoritative change.
   641	function reconcileFromSyncResponse(
   642	  sentBalance: number,
   643	  sentEpoch: number,
   644	  res: { zoomBalance: number; balanceEpoch: number; tonBalance?: number },
   645	  sentTonBalance?: number,
   646	): void {
   647	  // ORDER MATTERS — race fix.
   648	  //
   649	  // The naive ordering (bump _currentBalanceEpoch first, then dispatch the
   650	  // setState-driven balance snap) leaves a window where:
   651	  //   • _currentBalanceEpoch     == new (e.g. 6, just credited by the server)
   652	  //   • stateRef.current.balance == old (still 1000 — React hasn't committed
   653	  //                                       the snap setState yet)
   654	  // Any sync that fires inside that window (periodic doSync, tab-switch
   655	  // throttled doSync, or an immediate sync triggered by a stray tap) reads
   656	  // stateRef.current.balance == 1000 and ce == 6, then sends them to the
   657	  // server. The server's CASE WHEN balance_epoch > clientEpoch THEN keep
   658	  // ELSE GREATEST(0, client) END takes the ELSE branch (epoch == ce, not >),
   659	  // and overwrites the freshly credited 1100 back to 1000 — silently
   660	  // losing the wheel/admin/marketplace prize. Symptom: the YOU WON popup
   661	  // appears but the visible balance never rises.
   662	  //
   663	  // Fix: snap stateRef + _lastSyncedBalance SYNCHRONOUSLY, then bump the
   664	  // epoch. Now any concurrent sync sees the already-snapped (balance, epoch)
   665	  // pair and the server preserves the credit.
   666	  const serverAdvanced = res.balanceEpoch > sentEpoch;
   667	  const valueDiverged = res.zoomBalance !== sentBalance;
   668	  if (serverAdvanced && valueDiverged) {
   669	    if (_stateRefHolder) {
   670	      _stateRefHolder.current = { ..._stateRefHolder.current, balance: res.zoomBalance };
   671	    }
   672	    _lastSyncedBalance = res.zoomBalance;
   673	    _pendingSyncBalance = -1;
   674	    try {
   675	      window.dispatchEvent(new CustomEvent("zoom-server-balance-snap", {
   676	        detail: { balance: res.zoomBalance, epoch: res.balanceEpoch },
   677	      }));
   678	    } catch { /**/ }
   679	  }
   680	  // For TON we use a non-destructive merge on the server (GREATEST), so the
   681	  // server can return a value HIGHER than what we sent even when the epoch
   682	  // didn't advance (e.g. an earlier session credited TON, or an admin grant
   683	  // bumped the stored balance). Whenever the server reports a strictly
   684	  // higher TON than the client sent, snap local up so the user actually
   685	  // sees the credited amount. Same synchronous-stateRef-first ordering as
   686	  // the ZOOM snap above, for the same race-window reason.
   687	  if (
   688	    typeof res.tonBalance === "number" &&
   689	    typeof sentTonBalance === "number" &&
   690	    (res.tonBalance ?? 0) - (sentTonBalance ?? 0) > 1e-9
   691	  ) {
   692	    if (_stateRefHolder) {
   693	      _stateRefHolder.current = { ..._stateRefHolder.current, tonBalance: res.tonBalance };
   694	    }
   695	    _lastSyncedTonBalance = res.tonBalance;
   696	    try {
   697	      window.dispatchEvent(new CustomEvent("zoom-server-ton-snap", {
   698	        detail: { tonBalance: res.tonBalance, epoch: res.balanceEpoch },
   699	      }));
   700	    } catch { /**/ }
   701	  }
   702	  // Bump the epoch LAST so any sync that fires after this point already sees
   703	  // the snapped balance/TON in stateRef + _lastSyncedBalance.
   704	  setCurrentBalanceEpoch(res.balanceEpoch);
   705	}

   707	function immediateSyncToServer(state: GameState) {
   708	  const { telegramId } = getTelegramContext();
   709	  if (!telegramId) return;
   710	  const balance = Math.floor(state.balance);
   711	  const tonNow = Math.max(0, state.tonBalance || 0);
   712	  // Sync if EITHER currency changed since the last sync — TON-only changes
   713	  // (collect/reactivate of white planets) must persist promptly too.
   714	  const tonChanged = Math.abs(tonNow - _lastSyncedTonBalance) > 1e-9;
   715	  if (balance === _lastSyncedBalance && !tonChanged) return;
   716	
   717	  if (_syncInFlight) {
   718	    _pendingSyncBalance = balance;
   719	    return;
   720	  }
   721	
   722	  _lastSyncedBalance = balance;
   723	  _lastSyncedTonBalance = tonNow;
   724	  _syncInFlight = true;
   725	  const ctx_ = getTelegramContext();
   726	  const firstName = ctx_.firstName;
   727	  const username = ctx_.username;
   728	  const sentEpoch = _currentBalanceEpoch;
   729	  const sentTon = Math.max(0, state.tonBalance || 0);
   730	  syncBalance({ telegramId, firstName, username, zoomBalance: balance, tonBalance: sentTon, clientEpoch: sentEpoch })
   731	    .then((res) => {
   732	      reconcileFromSyncResponse(balance, sentEpoch, res, sentTon);
   733	      _syncInFlight = false;
   734	      if (_pendingSyncBalance >= 0 && _pendingSyncBalance !== _lastSyncedBalance) {
   735	        const nextBalance = _pendingSyncBalance;
   736	        _pendingSyncBalance = -1;
   737	        const { telegramId: tid, firstName: fn, username: un } = getTelegramContext();
   738	        if (tid) {
   739	          _lastSyncedBalance = nextBalance;
   740	          _syncInFlight = true;
   741	          const sentEpoch2 = _currentBalanceEpoch;
   742	          // Re-send the same TON value we just sent: this follow-up sync is
   743	          // only chasing the deferred ZOOM update; tonBalance state is owned
   744	          // by setState callbacks and we don't have access to it here.
   745	          const sentTon2 = sentTon;
   746	          syncBalance({ telegramId: tid, firstName: fn, username: un, zoomBalance: nextBalance, tonBalance: sentTon2, clientEpoch: sentEpoch2 })
   747	            .then((r2) => { reconcileFromSyncResponse(nextBalance, sentEpoch2, r2, sentTon2); _syncInFlight = false; })
   748	            .catch(() => { _syncInFlight = false; });
   749	        }
   750	      }
   751	    })
   752	    .catch(() => { _syncInFlight = false; });
   753	}

   833	// Parse the (kind, bundleIndex, subIndex) tuple out of a White or Earth
   834	// collection planet id. Returns null for any other planet id (regular,
   835	// bonus, marketplace, etc.). Format produced by makeWhiteCollectionPlanets
   836	// / makeEarthCollectionPlanets:
   837	//   `${kind}-${type}-b${bundleIndex}-${now}-${i}-${random}`
   838	//
   839	// `now` is `serverNow()` which can be a float (server-time offset includes
   840	// half-RTT calibration), so the timestamp segment must allow `.<digits>`.
   841	export function parseCollectionPlanetKey(
   842	  id: string,
   843	): { kind: "white" | "earth"; bundleIndex: number; subIndex: number } | null {
   844	  const m = /^(white|earth)-[A-Z0-9]+-b(\d+)-\d+(?:\.\d+)?-(\d+)-/.exec(id);
   845	  if (!m) return null;
   846	  return {
   847	    kind: m[1] as "white" | "earth",
   848	    bundleIndex: parseInt(m[2]!, 10),
   849	    subIndex: parseInt(m[3]!, 10),
   850	  };
   851	}

   874	function applyServerOverrides(
   875	  planets: Planet[],
   876	  serverByKey: Map<string, CollectionPlanetState>,
   877	): Planet[] {

   883	    if (!sp) return p;

   885	      ...p,
   886	      slotIndex: sp.slotIndex ?? null,
   887	      isFarmingActive: sp.isFarmingActive,
   888	      farmStartedAt: sp.farmStartedAtMs,
   889	      lastCollectedAt: sp.lastCollectedAtMs,

   938	/**
   939	 * Hard cutoff for the legacy never-started migration. Planets created on or
   940	 * after this instant always use the new init scheme (farmStartedAt = 0), so
   941	 * the migration is irrelevant to them and we refuse to touch them. Anything
   942	 * created before this instant predates the fix and is eligible for the
   943	 * migration check below.
   944	 *
   945	 * Set to the deploy moment of this fix (April 27, 2026 UTC). This bound is
   946	 * what eliminates any theoretical risk of misclassifying a planet that was
   947	 * (somehow) started in the same millisecond as its creation under the old
   948	 * code — such a planet, by definition, was created before the cutoff, but
   949	 * also we further require strict timestamp equality below, and the cutoff
   950	 * guarantees the migration cannot run forever / on future planets.
   951	 */
   952	const LEGACY_PLANET_MIGRATION_CUTOFF_MS = Date.UTC(2026, 3, 27, 0, 0, 0);
   953	
   954	/**
   955	 * One-time migration for legacy planets stored with farmStartedAt = createdAt.
   956	 *
   957	 * Before the cooldown-reset fix, every newly created planet (craft, bonus
   958	 * grant, buyer copy) was initialized with farmStartedAt = lastCollectedAt =
   959	 * createdAt = now, AND startFarming reset both timestamps to "now" on every
   960	 * call. After the fix, never-started planets must start at 0 so that the
   961	 * very first START opens a fresh 24h cycle. Legacy planets already in the
   962	 * user's local/server snapshot would otherwise be misclassified as
   963	 * "mid-cycle, just paused" and either resume from craft time or, if more
   964	 * than 24h have passed since craft, demand a reactivation fee for a cycle
   965	 * the user never actually got to use.
   966	 *
   967	 * The detection is conservative on multiple axes:
   968	 *   - Active or listed planets are skipped (they are clearly in use).
   969	 *   - farmStartedAt must be > 0 (new planets already use 0).
   970	 *   - farmStartedAt and lastCollectedAt must both exactly equal createdAt
   971	 *     (the unique fingerprint of the legacy "just-crafted-never-started"
   972	 *     state under the old init code).
   973	 *   - createdAt must be strictly before the deploy cutoff. Combined with
   974	 *     the inits-as-zero rule, this guarantees no future planet can ever
   975	 *     match the migration fingerprint, so the migration ages out naturally.
   976	 *
   977	 * After the very first START under the new code, farmStartedAt no longer
   978	 * equals createdAt (start time > craft time), so the migration self-
   979	 * disables for that planet too.
   980	 *
   981	 * Documented residual ambiguity (accepted tradeoff):
   982	 *   The migration cannot mathematically distinguish a true never-started
   983	 *   pre-cutoff planet from one that was started in the same millisecond as
   984	 *   its creation under the old code. A "false positive" here would gift a
   985	 *   single fresh 24h cycle to a single legacy planet. We accept this for
   986	 *   two reasons: (a) sub-millisecond human reaction time is physically
   987	 *   impossible (>16ms render frames, ~100ms minimum human reaction), and
   988	 *   the craft → render → tap pipeline forces multiple ticks between craft
   989	 *   time and any START click, so in practice no real planet ever has
   990	 *   farmStartedAt === createdAt unless it was truly never started; (b) the
   991	 *   alternative (no migration) charges real users a reactivation fee in
   992	 *   TON for cycles they never actually used, which is a far worse
   993	 *   real-money outcome than the theoretical false positive.
   994	 */
   995	function migrateLegacyNeverStartedPlanet<T extends Planet>(p: T): T {
   996	  if (p.isFarmingActive) return p;
   997	  if (p.isListedInMarket) return p;
   998	  if (p.farmStartedAt <= 0) return p;
   999	  if (p.farmStartedAt !== p.createdAt) return p;
  1000	  if (p.lastCollectedAt !== p.createdAt) return p;
  1001	  if (p.createdAt >= LEGACY_PLANET_MIGRATION_CUTOFF_MS) return p;
  1002	  return { ...p, farmStartedAt: 0, lastCollectedAt: 0 };
  1003	}

  1043	 * lastCollectedAt, capped to the 24h DAILY_COLLECT_MS window). Used by the UI
  1044	 * to show a live-ticking TON balance in the Pixel-Avatar modal.
  1045	 */
  1046	export function getWhitePlanetPendingTon(planet: Planet, now: number = serverNow()): number {
  1047	  if (planet.slotIndex == null || !planet.isFarmingActive) return 0;
  1048	  const cfg = PLANET_CONFIG[planet.name];
  1049	  if (!cfg.isTonFarming) return 0;
  1050	  // Defensive: if a TON-farming planet somehow lingers in state after the
  1051	  // admin revoked the underlying collection, we still don't credit anything.
  1052	  // Callers should also strip the planets from state, but this guard ensures
  1053	  // the live balance display can never re-credit a revoked collection.
  1054	  if (cfg.isTonFarming && !planet.farmStartedAt) return 0;
  1055	  const start = Math.max(planet.farmStartedAt, planet.lastCollectedAt);
  1056	  const end = Math.min(now, planet.farmStartedAt + FARM_DURATION_MS, planet.lastCollectedAt + DAILY_COLLECT_MS);
  1057	  if (end <= start) return 0;
  1058	  return (cfg.rate / 3_600_000) * (end - start);
  1059	}

  1061	/**
  1062	 * SUN cycle (24h) has elapsed since the last activation and a $ZOOM
  1063	 * reactivation fee is required to start a new cycle.
  1064	 */
  1065	export function isSunExpired(sun: SunState | null): boolean {
  1066	  if (!sun?.isOwned) return false;
  1067	  if (sun.farmStartedAt <= 0) return false;
  1068	  return serverNow() - sun.farmStartedAt > FARM_DURATION_MS;
  1069	}
  1070	
  1071	// Reactivation fee scales with how many SUNs the user owns: each SUN multiplies
  1072	// the per-cycle yield (1000/hr * sunCount), so each SUN must also pay its share
  1073	// of the fee — otherwise multi-SUN owners would reactivate for a fraction of
  1074	// what they earn. With sunCount=1 this is the historical 12,000 ZOOM.
  1075	// If the user owns no SUN, the fee is 0 (nothing to reactivate).
  1076	export function getSunReactivationFee(sunCount: number = 1): number {
  1077	  const n = Math.max(0, sunCount || 0);
  1078	  if (n <= 0) return 0;
  1079	  return SUN_CONFIG.reactivationFee * n;
  1080	}

  1096	export function formatDuration(ms: number): string {
  1097	  const s = Math.floor(ms / 1000);
  1098	  const d = Math.floor(s / 86400);
  1099	  const h = Math.floor((s % 86400) / 3600);
  1100	  const m = Math.floor((s % 3600) / 60);
  1101	  if (d > 0) return `${d}d ${h}h`;
  1102	  if (h > 0) return `${h}h ${m}m`;
  1103	  return `${m}m`;
  1104	}

  1109	function settleFarmingState(state: GameState, now: number): GameState {
  1110	  const from = state.lastFarmingSettledAt || now;
  1111	  if (now <= from) return state;
  1112	
  1113	  const speedMultiplier = 1 + (state.referralSpeedBonus || 0);
  1114	  let earned = 0;
  1115	
  1116	  for (const planet of state.planets) {
  1117	    if (!planet.isFarmingActive || planet.isListedInMarket) continue;
  1118	    const start = Math.max(from, planet.farmStartedAt, planet.lastCollectedAt);
  1119	    const end = Math.min(now, planet.farmStartedAt + FARM_DURATION_MS, planet.lastCollectedAt + DAILY_COLLECT_MS);
  1120	    if (end > start) {
  1121	      const dynamicRate = planet.rate + Math.random() * DYNAMIC_BONUS_MAX;
  1122	      earned += (dynamicRate / 3_600_000) * (end - start) * speedMultiplier;
  1123	    }
  1124	  }
  1125	
  1126	  // White Collection planets earn TON (not ZOOM) and accumulate into tonBalance
  1127	  // when the user presses COLLECT. Real-time pending TON for display is computed
  1128	  // separately via getWhitePlanetPendingTon(); here we only need to update the
  1129	  // settle timestamp — actual TON crediting happens on collectWhitePlanet().
  1130	
  1131	  if (state.sun?.isActive) {
  1132	    const start = Math.max(from, state.sun.farmStartedAt, state.sun.lastCollectedAt);
  1133	    const end = Math.min(now, state.sun.farmStartedAt + FARM_DURATION_MS, state.sun.lastCollectedAt + DAILY_COLLECT_MS);
  1134	    if (end > start) {
  1135	      const sunMultiplier = Math.max(1, state.sunCount || 1);
  1136	      earned += (SUN_CONFIG.rate * sunMultiplier / 3_600_000) * (end - start) * speedMultiplier;
  1137	    }
  1138	  }
  1139	
  1140	  if (earned <= 0) return { ...state, lastFarmingSettledAt: now };
  1141	
  1142	  return {
  1143	    ...state,
  1144	    balance: state.balance + earned,
  1145	    totalEarned: state.totalEarned + earned,
  1146	    seasonPoolEarned: state.seasonPoolEarned + earned,
  1147	    lastFarmingSettledAt: now,
  1148	  };
  1149	}

  2964	  // SPACE MERCHANT helpers — burn N planets of a given type, mint a
  2965	  // freshly-crafted (non-bonus) planet on success. Bonus planets that
  2966	  // get burned must call notifyPlanetBurn to keep the server-side
  2967	  // entitlement counter in lockstep, exactly like burnPlanet does.
  2968	  // ─────────────────────────────────────────────────────────────────
  2969	  const burnTwoOfType = useCallback((type: PlanetType): { ok: boolean; reason?: string } => {
  2970	    let outcome: { ok: boolean; reason?: string } = { ok: true };
  2971	    setState((prev) => {
  2972	      // Pick 2 idle, non-listed planets of the requested rarity. Prefer
  2973	      // non-bonus first so we don't drain server-granted entitlements
  2974	      // when the player has crafted alternatives available.
  2975	      const candidates = prev.planets.filter(
  2976	        (p) => p.name === type && !p.isFarmingActive && !p.isListedInMarket,
  2977	      );
  2978	      if (candidates.length < 2) {
  2979	        outcome = { ok: false, reason: `Need 2 idle ${PLANET_CONFIG[type].label} planets` };
  2980	        return prev;
  2981	      }
  2982	      const sorted = [...candidates].sort((a, b) => {
  2983	        const aBonus = a.id.startsWith(`bonus-${a.name}-`) ? 1 : 0;
  2984	        const bBonus = b.id.startsWith(`bonus-${b.name}-`) ? 1 : 0;
  2985	        return aBonus - bBonus; // non-bonus (0) first
  2986	      });
  2987	      const toBurn = sorted.slice(0, 2);
  2988	      const burnIds = new Set(toBurn.map((p) => p.id));
  2989	
  2990	      let cBasic = prev.claimedBonusBasic;
  2991	      let cRare = prev.claimedBonusRare;
  2992	      let cEpic = prev.claimedBonusEpic;
  2993	      let cGold = prev.claimedBonusGold;
  2994	      for (const p of toBurn) {
  2995	        const isBonus = p.id.startsWith(`bonus-${p.name}-`);
  2996	        if (isBonus && prev.telegramId && (p.name === "BASIC" || p.name === "RARE" || p.name === "EPIC" || p.name === "GOLD")) {
  2997	          notifyPlanetBurn(prev.telegramId, p.name);
  2998	          if (p.name === "BASIC") cBasic = Math.max(0, cBasic - 1);
  2999	          else if (p.name === "RARE") cRare = Math.max(0, cRare - 1);
  3000	          else if (p.name === "EPIC") cEpic = Math.max(0, cEpic - 1);
  3001	          else if (p.name === "GOLD") cGold = Math.max(0, cGold - 1);
  3002	        }
  3003	      }
  3004	
  3005	      const updated: GameState = {
  3006	        ...prev,
  3007	        planets: prev.planets.filter((p) => !burnIds.has(p.id)),
  3008	        claimedBonusBasic: cBasic,
  3009	        claimedBonusRare: cRare,
  3010	        claimedBonusEpic: cEpic,
  3011	        claimedBonusGold: cGold,
  3012	      };
  3013	      stateRef.current = updated;
  3014	      saveState(updated);
  3015	      return updated;
  3016	    });
  3017	    return outcome;
  3018	  }, []);
